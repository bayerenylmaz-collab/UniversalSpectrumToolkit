\begin{frame}{Bu sunumda ne anlatılacak?}

%---------------------------------------------------------------------
\begin{frame}{Sorun neydi?}
  \begin{alertblock}{Kısa cevap}
    ROOT bir analiz çerçevesidir; \texttt{.spe} / \texttt{.chn} ise cihaz/MCA spektrum formatıdır.
    ROOT bu dosyayı \textbf{native spektrum nesnesi} olarak bilmez.
  \end{alertblock}

  \vspace{0.6em}
  \textbf{Pratik sonuç:}
  \begin{itemize}
    \item Dosyayı ``açmak'' ile spektrumu ``doğru okumak'' aynı şey değildir
    \item Histogram kendiliğinden oluşmaz
    \item Kör binary tahmin gürültü / saçma değerler üretebilir
  \end{itemize}
\end{frame}

\begin{frame}{Neden? Biraz daha teknik}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{ROOT ne bekler?}
      \begin{itemize}
        \item \texttt{.root} dosyası
        \item \texttt{TH1}, \texttt{TTree} gibi nesneler
        \item Yerleşik I/O ile tanımlı biçimler
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\texttt{.spe} ise ne?}
      \begin{itemize}
        \item Ham bayt dizisi
        \item Tek bir uluslararası standart değil
        \item ASCII Ortec, GF3 binary, ...
        \item Header + kanal count yapısı vendor'a göre değişir
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{1em}
  \begin{block}{Özet}
    ROOT spektrumu ``görmez''; çünkü dosyanın anlamını bilmez.
    Bizim makro bu anlamı tanıyan \textbf{çevirici katmandır}.
  \end{block}
\end{frame}

\begin{frame}{Makro nasıl üstesinden geliyor?}
  \begin{enumerate}
    \item Dosyayı ham bayt olarak okur
    \item Uzantıya uygun kayıtlı okuyucuları dener
    \item Eşleşen formatı seçer (ASCII SPE / GF3-like / CHN)
    \item \textbf{kanal $\rightarrow$ counts} dizisini çıkarır
    \item ROOT'un anladığı nesnelere yazar:
      \begin{itemize}
        \item \texttt{TH1D hSpectrumChannel}
        \item \texttt{TTree spectrum}
        \item \texttt{conversion\_metadata}
        \item ham baytlar (\texttt{raw/raw\_bytes})
      \end{itemize}
  \end{enumerate}

  \vspace{0.4em}
  \begin{exampleblock}{Kritik ilke}
    Bilinmeyen formatı sessizce uydurmaz. Önce tanır, emin değilse raporlar.
  \end{exampleblock}
\end{frame}

\begin{frame}{Makro yokken: kör tahmin}
  \begin{alertblock}{Ne oluyor?}
    Binary dosya spektrum formatı bilinmeden \texttt{uint16} veya \texttt{float32}
    sanılırsa sonuç fiziksel olarak anlamsızlaşır.
  \end{alertblock}

  \vspace{0.5em}
  \begin{itemize}
    \item Yoğun gürültü zemini
    \item Rastgele dikenler
    \item $10^{18}$ mertebesinde saçma değerler
  \end{itemize}

  \vspace{0.6em}
  \textcolor{ustgray}{\small Bu yüzden ``ROOT ile bir şekilde çizildi'' demek yeterli değildir; doğru layout şarttır.}
\end{frame}

\begin{frame}{Gerçek kullanımda ne görüyoruz?}
  \textbf{Örnek:} Eu-152 GF3-like binary SPE

  \vspace{0.5em}
  \begin{itemize}
    \item Dosya boyutu: $\sim$65\,KB
    \item Okuyucu kararı: \textbf{GF3-like binary SPE MATCH}
    \item Kanal sayısı: 16384
    \item Tip: \texttt{float32}, little-endian
    \item Enerji kalibrasyonu: yok (dosyada taşınmıyorsa)
    \item Çıktı: aynı isimli \texttt{.root} + histogram
  \end{itemize}

  \vspace{0.7em}
  \begin{exampleblock}{Ekip testi}
    Eu-152 ve Co-60 verilerinde spektrum şekli ve pik yapıları
    beklenen fizikle uyumlu bulundu.
  \end{exampleblock}
\end{frame}

\begin{frame}{Ne okur / ne okumaz?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Güvenilir}
        \begin{itemize}
          \item Ortec ASCII SPE
          \item GF3-like binary SPE
          \item Ortec binary CHN
        \end{itemize}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Otomatik garanti yok}
        \begin{itemize}
          \item Her vendor'ın her SPE türevi
          \item Bozuk / eksik header
          \item Tanımsız binary layout
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.8em}
  \begin{center}
    \textbf{Doğru ifade:} ``Laboratuvarda sık görülen SPE/CHN için genelde okur.''\\
    \textcolor{ustgray}{``Dünyadaki her SPE'' demek doğru olmaz.}
  \end{center}
\end{frame}
